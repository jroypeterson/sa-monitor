   137	    r"|\bfda\s+has\s+granted\s+approval\b"
   138	    r"|\bapproved\s+by\s+the\s+fda\b",
   139	    re.I,
   140	)
   141	# HARD-pending / procedural milestones that are NOT a grant. Any hit suppresses
   142	# BOTH approval and clearance. NOTE (round-2 #4): designation words — priority
   143	# review, fast track, breakthrough therapy, orphan drug, rare pediatric — are
   144	# DELIBERATELY NOT here. They must not veto a real approval verb ("FDA approves
   145	# X under Priority Review"), and they aren't needed as suppressors: "FDA grants
   146	# Priority Review to X" carries no approve/clear verb, so _APPROVAL/_CLEARANCE
   147	# never match it → still None.
   148	_APPROVAL_NEGATIVE = re.compile(
   149	    r"\baccept(?:s|ed)?\b"
   150	    r"|\badvisory\s+committee\b|\badcomm\b|\bpdufa\b"
   151	    r"|\bresubmi|\bseeking\b|\bawait|\bapplication\s+for\b"
   152	    r"|\bfiles?\s+for\b|\bsubmit|\bsubmission\b|\bto\s+request\b"
   153	    r"|\bcomplete\s+response\s+letter\b|\bpre-?market\s+notification\b",
   154	    re.I,
   155	)
   156	
   157	# Future / intent framing toward a grant that hasn't happened yet (round-2 #2):
   158	# "targets/expects/pending/plans to/seeks/on track for FDA clearance", and the
   159	# reverse "FDA clearance expected/pending". Anchored to the grant noun (with a
   160	# whitelisted determiner/adjective window) so an approval that merely CONTAINS
   161	# "expected" ("approved sooner than expected", "better than expected") is NOT
   162	# suppressed. Fillers are a closed set of determiners/adjectives, never verbs,
   163	# so "plans to commercialize following FDA approval" (a real approval) doesn't
   164	# get vetoed.
   165	_REG_FUTURE_INTENT = re.compile(
   166	    # intent-then-noun
   167	    r"\b(?:pending|expect\w*|anticipat\w*|target\w*|plan\w*\s+to|intends?\s+to"
   168	    r"|seek\w*|on\s+track\s+(?:for|to))\s+"
   169	    r"(?:(?:the|a|an|its|potential|possible|marketing|regulatory|full|final|us|u\.s\.)\s+){0,2}"
   170	    r"(?:fda\s+)?(?:approval|clearance|authorization|de\s+novo)\b"
   171	    r"|"
   172	    # noun-then-intent
   173	    r"\b(?:fda\s+)?(?:approval|clearance|authorization)\b\s+"
   174	    r"(?:(?:is|remains|still|now|currently)\s+){0,2}"
   175	    r"(?:pending|expect\w*|anticipat\w*|target\w*|on\s+track)\b",
   176	    re.I,
   177	)
   178	
   179	# Denial / rejection grammar (F2). A rejection must NEVER read as a grant. v1
   180	# has no "rejection" event type, so a plain "did not receive approval" headline
   181	# → None (missing it is far safer than inverting it to an approval). Gates BOTH
   182	# approval and clearance. NOT applied to trial readouts (which have their own
   183	# missed-endpoint direction) or to CRL (the rejection event we DO fire).
   184	_REG_DENIAL = re.compile(
   185	    r"\bdid\s+not\b|\bdoes\s+not\b|\bdid\s*n[’']?t\b|\bdoes\s*n[’']?t\b"
   186	    r"|\bfailed\s+to\b|\bfails\s+to\b"
   187	    r"|\bnot\s+(?:receive|receiv\w+|win|won|gain\w*|secure\w*|grant\w*|approv\w+|clear\w+)\b"
   188	    r"|\bdenie[sd]\b|\bdenial\b|\bdeclin\w*\s+to\s+approve\b"
   189	    r"|\breject(?:s|ed|ion)?\b|\bnot\s+approved\b|\bunapproved\b"
   190	    r"|\bwithdraw\w*\b|\brescind\w*\b|\brevoke[sd]?\b",
   191	    re.I,
   192	)
   193	
   194	# Commentary / competitor framing (F3): an issuer reacting to SOMEONE ELSE's
   195	# FDA action. _emit_hc_events would mis-attribute it to the commenting issuer,
   196	# so suppress. Accept the rare miss of a self-approval PR that also contains one
   197	# of these words — precision-first.
   198	_REG_COMMENTARY = re.compile(
   199	    r"\bcomment(?:s|ary|ing|ed)?\s+on\b|\bcommentary\b"
   200	    r"|\bcongratulat\w+|\bapplaud\w*|\bwelcome[sd]?\b|\breact[s]?\s+to\b"
   201	    r"|\brespond[s]?\s+to\b|\bissues?\s+statement\b"
   202	    r"|\bstatement\s+(?:on|regarding|re)\b"
   203	    # round-2 #3: generic commentary verbs, anchored to an FDA-action noun so a
   204	    # self-approval PR that merely says "announces … and notes X" still fires
   205	    # (announce is not commentary; only these verbs pointed AT the action are).
   206	    r"|\b(?:highlights?|notes?|discuss\w*|points?\s+to|cite[sd]?|references?)\b"
   207	    r"(?:\W+\w+){0,3}?\W+(?:fda\s+)?(?:approval|clearance|authorization)\b",
   208	    re.I,
   209	)
   210	
   211	# CRL (F5). Must NOT match bare \bcrl\b — CRL is also the ticker for Charles
   212	# River Laboratories, a COVERED name ("CRL Announces … Financial Results").
   213	# Require the full phrase or a regulatory-CRL idiom.
   214	_CRL = re.compile(
   215	    r"\bcomplete\s+response\s+letter\b"
   216	    r"|\breceiv\w*\s+(?:a\s+)?crl\b"
   217	    r"|\bcrl\s+from\s+(?:the\s+)?fda\b"
   218	    r"|\bissue[sd]?\s+(?:a\s+)?crl\b"
   219	    r"|\bfda\s+issue[sd]?\s+(?:a\s+)?crl\b",
   220	    re.I,
   221	)
   222	
   223	# Medtech clearance. Requires a completed clearance action — bare adjective
   224	# "FDA-cleared" (e.g. "operates an FDA-cleared facility") must NOT match.
   225	_CLEARANCE = re.compile(
   226	    r"\bfda\s+clears\b"
   227	    r"|\bfda\s+clearance\b"
   228	    r"|\breceive[sd]?\s+fda\s+clearance\b"
   229	    r"|\b510\(k\)\s+clearance\b"
   230	    r"|\bgranted\s+510\(k\)\b"
   231	    r"|\bde\s+novo\s+(?:clearance|authorization|grant)\b"
   232	    r"|\bpma\s+approval\b"
   233	    r"|\bfda\s+grants?\s+(?:de\s+novo|510)\b",
   234	    re.I,
   235	)

   238	def _reg_suppressed(title: str) -> bool:
   239	    """A grant (approval or clearance) is suppressed when the headline is
   240	    actually a hard-pending milestone, a future/intent framing toward the grant,
   241	    a denial/rejection, or third-party commentary."""
   242	    return bool(
   243	        _APPROVAL_NEGATIVE.search(title)
   244	        or _REG_FUTURE_INTENT.search(title)
   245	        or _REG_DENIAL.search(title)
   246	        or _REG_COMMENTARY.search(title)
   247	    )

   249	
   250	def _classify_regulatory(title: str, item: NewsItem) -> Optional[HCEvent]:
   251	    # CRL first — a Complete Response Letter is unambiguous and distinct from
   252	    # an approval; it must never be mis-read as one.
   253	    if _CRL.search(title):
   254	        return _build(item, CRL, DIR_NA)
   255	    # Approval + clearance share the same suppression gate: pending milestone,
   256	    # denial/rejection (F2), or third-party commentary (F3). Clearance was
   257	    # previously ungated (F1).
   258	    if _reg_suppressed(title):
   259	        return None
   260	    if _APPROVAL.search(title):
   261	        return _build(item, FDA_APPROVAL, DIR_NA)
   262	    if _CLEARANCE.search(title):
   263	        return _build(item, CLEARANCE, DIR_NA)
   264	    return None
   265

   269	# --------------------------------------------------------------------------

   272	def _build(item: NewsItem, event_type: str, direction: str,
   273	           *, phase: str = "") -> HCEvent:
   274	    return HCEvent(
   275	        news_id=item.news_id,
   276	        symbol="",  # attribution decided by the emit path (tickers ∩ universe)
   277	        event_type=event_type,
   278	        direction=direction,
   279	        headline=item.title,
   280	        source=item.source,
   281	        url=item.url,
   282	        published_at=item.published_at,
   283	        phase=phase,
   284	        confidence="high",
   285	        raw_industries=tuple(item.industries),
   286	    )

   289	def classify(item: NewsItem) -> Optional[HCEvent]:
   290	    """Classify a NewsItem into an HCEvent, or None if it isn't one.
   291	
   292	    Precision-biased: returns None on any ambiguity. Regulatory actions
   293	    (approval / CRL / clearance) are checked before trial readouts — an
   294	    approval PR carries no Phase grammar, so the two families don't collide.
   295	    """
   296	    title = item.title or ""
   297	    if not title.strip():
   298	        return None
   299	
   300	    reg = _classify_regulatory(title, item)
   301	    if reg is not None:
   302	        return reg
   303	
   304	    return _classify_trial(title, item)
