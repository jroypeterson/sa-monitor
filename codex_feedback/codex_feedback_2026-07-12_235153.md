   142	def _halt_dt_to_utc(halt: HaltEvent) -> Optional[datetime]:
   143	    """Convert halt_date + halt_time (in ET) to a UTC datetime.
   144	
   145	    Halt feeds publish wall-clock ET time; converting to UTC requires DST
   146	    awareness. Use zoneinfo's America/New_York which handles EDT/EST splits
   147	    correctly. Returns None on malformed input rather than raising.
   148	    """
   149	    if not halt.halt_date or not halt.halt_time:
   150	        return None
   151	    parts = halt.halt_time.split(":")
   152	    if len(parts) < 2:
   153	        return None
   154	    try:
   155	        y, m, d = (int(x) for x in halt.halt_date.split("-"))
   156	        hh = int(parts[0])
   157	        mm = int(parts[1])
   158	        ss = int(float(parts[2])) if len(parts) >= 3 else 0
   159	    except (ValueError, IndexError):
   160	        return None
   161	    et = ZoneInfo("America/New_York")
   162	    try:
   163	        local = datetime(y, m, d, hh, mm, ss, tzinfo=et)
   164	    except ValueError:
   165	        return None
   166	    return local.astimezone(timezone.utc)
